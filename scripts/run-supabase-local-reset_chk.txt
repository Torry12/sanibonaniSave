param(
    [string]$SeedFile,
    [switch]$NoSeed,
    [switch]$SkipStart,
    [switch]$CheckOnly,
    [ValidateSet("Light", "Extensive", "None")]
    [string]$SeedingMode = "Light"
)

$ErrorActionPreference = "Stop"

function Invoke-Step {
    param(
        [Parameter(Mandatory = $true)]
        [string]$Command
    )

    Write-Host "> $Command" -ForegroundColor Cyan
    Invoke-Expression $Command
    if ($LASTEXITCODE -ne 0) {
        throw "Command failed with exit code ${LASTEXITCODE}: $Command"
    }
}

$repoRoot = Split-Path -Parent $PSScriptRoot
Set-Location $repoRoot

Write-Host "Repo: $repoRoot" -ForegroundColor Green

if (-not (Get-Command supabase -ErrorAction SilentlyContinue)) {
    throw "Supabase CLI not found. Install from https://github.com/supabase/cli"
}

Invoke-Step "supabase --version"

if ($CheckOnly) {
    Write-Host "Check-only mode complete." -ForegroundColor Yellow
    exit 0
}

if (-not $SkipStart) {
    Invoke-Step "supabase start"
}

if ($NoSeed) {
    Invoke-Step "supabase db reset --local --no-seed --yes"
} else {
    Invoke-Step "supabase db reset --local --no-seed --yes"
}

# Find DB container name
$dbContainer = docker ps --format "{{.Names}}" | Select-String "supabase_db" | Select-Object -First 1
if (-not $dbContainer) {
    throw "Supabase DB container not found. Is Supabase running?"
}
Write-Host "Using DB Container: $dbContainer" -ForegroundColor Yellow

# Apply core functions and logic not in standard migrations
Write-Host "Applying core functions and enhancements via psql..." -ForegroundColor Yellow
$coreScripts = @(
    "13_ADD_MISSING_RPCS.sql",
    "14_FIX_PLATFORM_FEE_RPC.sql",
    "16_LEDGER_SYSTEM_UPGRADE.sql",
    "17_ATOMIC_PAYMENT_PROCESSING.sql",
    "18_PAYOUT_RPCS.sql",
    "19_INTEGRITY_TRIGGERS.sql",
    "20_ALIGN_CONTRIBUTION_TYPES.sql",
    "21_FINANCIAL_ACCURACY_POLISH.sql",
    "22_CLAIM_RPCS.sql",
    "23_LOAN_INTEGRITY_TRIGGERS.sql",
    "24_GLOBAL_FINANCIAL_REPAIR.sql",
    "25_UPGRADE_LEDGER_MEMBER_TRACKING.sql",
    "26_BANK_INTEGRATION_SCHEMA.sql",
    "27_FINANCIAL_SYSTEM_HARDENING.sql",
    "33_LEDGER_CONSOLIDATION_V1.sql",
    "34_AUDIT_REPORT_FACILITIES_V1.sql",
    "36_LEDGER_REFACTOR_V2.sql",
    "37_ADD_CLAIM_DOCUMENTS.sql",
    "38_FIX_LEDGER_INTEGRITY_RPCS.sql"
)

foreach ($script in $coreScripts) {
    $scriptPath = "supabase/$script"
    if (Test-Path $scriptPath) {
        Write-Host "Applying $script..." -ForegroundColor Cyan
        Get-Content $scriptPath -Raw | docker exec -i $dbContainer psql -U postgres -d postgres -q
        if ($LASTEXITCODE -ne 0) {
            throw "Failed to apply $script"
        }
    }
}

if ($SeedingMode -ne "None") {
    Write-Host "Applying Seeding Mode: $SeedingMode..." -ForegroundColor Yellow
    if ($SeedingMode -eq "Extensive") {
        Write-Host "Seeding Extensive Data..."
        Get-Content "supabase/28_COMPREHENSIVE_SEED_DATA.sql" -Raw | docker exec -i $dbContainer psql -U postgres -d postgres -q
        Get-Content "supabase/35_SCENARIO_BASED_ENHANCEMENTS.sql" -Raw | docker exec -i $dbContainer psql -U postgres -d postgres -q
        Get-Content "supabase/31_REAL_LIFE_INTEGRITY_SEEDS.sql" -Raw | docker exec -i $dbContainer psql -U postgres -d postgres -q
    } else {
        # Light mode: just run standard seed
        Write-Host "Seeding Light Data..."
        Get-Content "supabase/06_SEED_DATA.sql" -Raw | docker exec -i $dbContainer psql -U postgres -d postgres -q
    }
}

Write-Host "Local migrations and seeding applied successfully." -ForegroundColor Green
Invoke-Step "supabase migration list --local"
